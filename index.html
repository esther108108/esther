<script>

/* ---------------------------------------------------------------

   ELEMENTS & GLOBAL STATE

----------------------------------------------------------------*/

const radios = document.querySelectorAll('[name="product"]');

const txt    = document.getElementById('engraveText');

const font   = document.getElementById('fontSelect');

const fSize  = document.getElementById('fontSize');

const view   = document.getElementById('viewCanvas');

const vCtx   = view.getContext('2d');

const full   = document.createElement('canvas');      // full-rez preview

const fCtx   = full.getContext('2d');

const genBtn = document.getElementById('generateBtn');

const viewportMeta = document.querySelector('meta[name="viewport"]');

 

let img=null, pos={x:0,y:0}, scale=1;

let dragging=false, offX=0, offY=0, activeId=null;

 

/* ---------------------------------------------------------------

   HELPERS

----------------------------------------------------------------*/

async function loadImageHighRes(src){

  const blob = await fetch(src,{mode:'cors'}).then(r=>r.blob());

  const url  = URL.createObjectURL(blob);

  return new Promise(res=>{

    const i=new Image();

    i.onload=()=>{URL.revokeObjectURL(url);res(i);};

    i.src=url;

  });

}

function updateScale(){

  const MAX_W=900, MAX_H=600;

  scale=Math.min(1,MAX_W/img.naturalWidth,MAX_H/img.naturalHeight);

  view.width = img.naturalWidth*scale;

  view.height= img.naturalHeight*scale;

  full.width = img.naturalWidth;

  full.height= img.naturalHeight;

}

function draw(){

  if(!img)return;

  /* full-rez */

  fCtx.clearRect(0,0,full.width,full.height);

  fCtx.drawImage(img,0,0);

  fCtx.font=`${fSize.value}px ${font.value}`; fCtx.fillStyle='#000'; fCtx.textBaseline='top';

  fCtx.fillText(txt.value,pos.x,pos.y);

  /* scaled view */

  vCtx.setTransform(scale,0,0,scale,0,0);

  vCtx.clearRect(0,0,view.width/scale,view.height/scale);

  vCtx.drawImage(img,0,0);

  vCtx.font=`${fSize.value}px ${font.value}`; vCtx.fillStyle='#000'; vCtx.textBaseline='top';

  vCtx.fillText(txt.value,pos.x,pos.y);

  vCtx.setTransform(1,0,0,1,0,0);

}

function textHit(mx,my){

  vCtx.font=`${fSize.value}px ${font.value}`;

  const w=vCtx.measureText(txt.value).width*scale;

  const h=parseInt(fSize.value,10)*scale;

  return mx>=pos.x*scale && mx<=pos.x*scale+w && my>=pos.y*scale && my<=pos.y*scale+h;

}

function downloadCanvas(can,filename){

  const trigger=url=>{

    const a=document.createElement('a');a.href=url;a.download=filename;

    document.body.appendChild(a);a.click();

    setTimeout(()=>{URL.revokeObjectURL(url);a.remove();},100);

  };

  can.toBlob?can.toBlob(b=>trigger(URL.createObjectURL(b))):trigger(can.toDataURL('image/png'));

}

 

/* ---------------------------------------------------------------

   DRAGGING  (pointer events + touch fallback)

----------------------------------------------------------------*/

function startDrag(x,y,id){

  const rect=view.getBoundingClientRect();

  const mx=x-rect.left, my=y-rect.top;

  if(!textHit(mx,my)) return;

  dragging=true; activeId=id;

  offX=mx-pos.x*scale; offY=my-pos.y*scale;

}

function moveDrag(x,y){

  if(!dragging) return;

  const rect=view.getBoundingClientRect();

  pos.x=(x-rect.left-offX)/scale;

  pos.y=(y-rect.top -offY)/scale;

  draw();

}

function endDrag(id){

  if(id!==undefined && id!==activeId) return;

  dragging=false; activeId=null;

}

 

/* pointer-event path */

if('PointerEvent' in window){

  view.addEventListener('pointerdown',e=>startDrag(e.clientX,e.clientY,e.pointerId));

  window.addEventListener('pointermove',e=>moveDrag(e.clientX,e.clientY));

  window.addEventListener('pointerup',e=>endDrag(e.pointerId));

  window.addEventListener('pointercancel',e=>endDrag(e.pointerId));

}

/* very old browsers: touch path */

else{

  view.addEventListener('touchstart',e=>{

    if(e.touches.length!==1) return;

    startDrag(e.touches[0].clientX,e.touches[0].clientY,0);

  },{passive:true});

  window.addEventListener('touchmove',e=>{

    if(!dragging) return;

    moveDrag(e.touches[0].clientX,e.touches[0].clientY);

  },{passive:true});

  window.addEventListener('touchend',()=>endDrag(0));

}

 

/* ---------------------------------------------------------------

   OTHER EVENTS

----------------------------------------------------------------*/

radios.forEach(r=>r.addEventListener('change',async e=>{

  img=await loadImageHighRes(e.target.value);

  updateScale();

  pos.x=img.naturalWidth*0.5-100;

  pos.y=img.naturalHeight*0.5-20;

  draw();

  genBtn.disabled=false;

}));

 

[txt,font,fSize].forEach(el=>el.addEventListener('input',draw));

 

/* viewport zoom-reset after typing */

txt.addEventListener('focus',()=>{

  viewportMeta.content='width=device-width,initial-scale=1,maximum-scale=10';

});

txt.addEventListener('blur',()=>{

  viewportMeta.content='width=device-width,initial-scale=1,maximum-scale=1';

  setTimeout(()=>window.scrollTo({top:0,left:0}),50);

});

 

/* EXPORT */

genBtn.addEventListener('click',()=>{

  draw();

  downloadCanvas(full,'preview.png');              // knife + text

  const tCan=document.createElement('canvas');     // text only

  tCan.width=full.width; tCan.height=full.height;

  const tCtx=tCan.getContext('2d');

  tCtx.font=`${fSize.value}px ${font.value}`; tCtx.fillStyle='#000'; tCtx.textBaseline='top';

  tCtx.fillText(txt.value,pos.x,pos.y);

  downloadCanvas(tCan,'text-only.png');

});

</script>
